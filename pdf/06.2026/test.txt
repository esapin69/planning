<script>
function formatMonth(date){
  const mm = String(date.getMonth() + 1).padStart(2, "0");
  const yyyy = date.getFullYear();
  return `${mm}.${yyyy}`;
}

function nextMonth(date){
  return new Date(date.getFullYear(), date.getMonth() + 1, 1);
}

const today = new Date();
const visibleMonths = [formatMonth(today)];

if (today.getDate() >= 20) {
  visibleMonths.push(formatMonth(nextMonth(today)));
}

const packsDiv = document.getElementById("packs");

visibleMonths.forEach(month=>{
  const div = document.createElement("div");
  div.className = "pack";

  div.innerHTML = `
    <img src="assets/mois/${month}.jpg" alt="">
    <button class="zone"></button>
  `;

  div.querySelector(".zone").onclick = ()=>{
    window.location.href = `pdf/${month}/`;
  };

  packsDiv.appendChild(div);
});
</script>
